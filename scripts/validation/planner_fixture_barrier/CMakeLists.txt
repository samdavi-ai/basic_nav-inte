cmake_minimum_required(VERSION 3.16)
project(vira_planner_fixture_barrier)
find_package(ament_cmake REQUIRED)
find_package(nav2_navfn_planner REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
add_library(${PROJECT_NAME} SHARED barrier.cpp clearance_navfn.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} nav2_navfn_planner nav2_core pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
find_package(dwb_core REQUIRED)
add_library(vira_swept_critic SHARED swept_critic.cpp policy_preference.cpp)
ament_target_dependencies(vira_swept_critic dwb_core pluginlib)
pluginlib_export_plugin_description_file(dwb_core critic.xml)
install(TARGETS vira_swept_critic LIBRARY DESTINATION lib)
ament_package()
